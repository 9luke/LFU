		name = "USA Fighter Shipments"

		icon = GFX_decision_generic_air

			original_tag = ENG
		}

		available = {
			hidden_trigger = { always = no }
		}

		visible = { 
			has_country_flag = no_asia_fighter_LL
		}

		is_good = yes

            AND = {
                has_country_flag = no_asia_fighter_LL

            }
		}

		timeout_effect = {
            if = { 
				limit = { date < 1940.1.1 }

					type = fighter_equipment_2
					amount = 250
					producer = ENG
                    design_team = ENG_supermarine_mio
				}	
			}
			else_if = { 
				limit = { 
					date > 1940.1.1
					date < 1940.6.1
				}
				add_equipment_to_stockpile = {
					type = fighter_equipment_2
					amount = 500
					producer = ENG
                    design_team = ENG_supermarine_mio
				}	
			}
			else_if = { 
				limit = { 
					date > 1940.6.1
				}
				add_equipment_to_stockpile = {
					type = fighter_equipment_2
					amount = 1000
					producer = ENG
                    design_team = ENG_supermarine_mio
				}
