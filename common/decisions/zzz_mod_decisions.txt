operations = {
    no_asia_uk_LL = {

		name = "USA Fighter Shipments"

		icon = GFX_decision_generic_air
			

		allowed = {
			original_tag = ENG
		}

		available = {
			hidden_trigger = { always = no }
		}

		visible = { 
			has_country_flag = no_asia_fighter_LL
		}

		is_good = yes
		activation = { 
            AND = {
                has_country_flag = no_asia_fighter_LL 
                NOT = { has_active_mission = no_asia_uk_LL_2 }
                has_tech = fighter2
            }
		}
		days_mission_timeout = 30

		cancel_if_not_visible = yes

		timeout_effect = {
            if = { 
				limit = { date < 1940.1.1 }
					add_equipment_to_stockpile = {
					type = fighter_equipment_2
					amount = 250
					producer = ENG
                    design_team = mio:ENG_supermarine_mio
				}	
			}
			else_if = { 
				limit = { 
					date > 1940.1.1
					date < 1940.6.1
				}
				add_equipment_to_stockpile = {
					type = fighter_equipment_2
					amount = 500
					producer = ENG
                    design_team = mio:ENG_supermarine_mio
				}	
			}
			else_if = { 
				limit = { 
					date > 1940.6.1
				}
				add_equipment_to_stockpile = {
					type = fighter_equipment_2
					amount = 1000
					producer = ENG
                    design_team = mio:ENG_supermarine_mio
				}	
			}
			
			activate_mission = no_asia_uk_LL_2
		}			
    }

	annex_hungary = {
		name = "Annex Hungary"
		fire_only_once = yes
		cost = 0

		available = {
			HUN = { is_ai = yes }
		}

		ai_will_do = {
			factor = 0
		}

		allowed = { tag = ROM }
		targets = { HUN }

		complete_effect = {
			HUN = {
				transfer_units_fraction = {
					target = ROM
					size = 1
					stockpile_ratio = 1
					army_ratio = 1
					navy_ratio = 1
					air_ratio = 1
					keep_unit_leaders_trigger = {
						always = no
					}
				}
			}
			ROM = {
				annex_country = {
					target = HUN
					transfer_troops = yes
				}
				if = {
					limit = { NOT = { has_idea = zhm_command_power } }
					add_ideas = zhm_command_power
				}
				set_cosmetic_tag = HUN_ROM_KINGDOM
			}
			every_state = {
				limit = {
					is_core_of = HUN
					is_controlled_by = ROM
				}
				add_core_of = ROM
				remove_core_of = HUN
			}
		}
	}
